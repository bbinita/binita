@api_view(['POST'])
def submit_complaint(request):
    category = classify(
        request.data['description']
    )
    serializer = ComplaintSerializer(
        data=request.data
    )
    if serializer.is_valid():
        serializer.save(
            user=request.user,
            category=category
        )
        return Response(
            serializer.data,
            status=201
        )
